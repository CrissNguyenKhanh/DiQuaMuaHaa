import mysql.connector
from mysql.connector import Error
import hashlib
import json
from datetime import datetime

class Database:
    def __init__(self):
        self.host = 'localhost'
        self.port = 3306
        self.user = 'root'
        self.password = ''  # XAMPP mặc định không có password
        self.database = 'medical_ai_db'
        
        # Tạo database nếu chưa có
        self.create_database()
        self.init_tables()
    
    def get_connection(self):
        """Kết nối đến MySQL"""
        try:
            connection = mysql.connector.connect(
                host=self.host,
                port=self.port,
                user=self.user,
                password=self.password,
                database=self.database
            )
            return connection
        except Error as e:
            print(f"❌ Lỗi kết nối MySQL: {e}")
            return None
    
    def create_database(self):
        """Tạo database nếu chưa tồn tại"""
        try:
            connection = mysql.connector.connect(
                host=self.host,
                port=self.port,
                user=self.user,
                password=self.password
            )
            cursor = connection.cursor()
            cursor.execute(f"CREATE DATABASE IF NOT EXISTS {self.database} CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci")
            print(f"✅ Database '{self.database}' đã sẵn sàng")
            cursor.close()
            connection.close()
        except Error as e:
            print(f"❌ Lỗi tạo database: {e}")
    
    def init_tables(self):
        """Khởi tạo các bảng"""
        connection = self.get_connection()
        if not connection:
            return
        
        cursor = connection.cursor()
        
        # Bảng users
        cursor.execute('''
            CREATE TABLE IF NOT EXISTS users (
                id INT AUTO_INCREMENT PRIMARY KEY,
                username VARCHAR(50) UNIQUE NOT NULL,
                password VARCHAR(255) NOT NULL,
                full_name VARCHAR(100),
                email VARCHAR(100),
                role VARCHAR(20) DEFAULT 'user',
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
            ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci
        ''')
        
        # Bảng medical_records
        cursor.execute('''
            CREATE TABLE IF NOT EXISTS medical_records (
                id INT AUTO_INCREMENT PRIMARY KEY,
                user_id INT NOT NULL,
                symptoms TEXT NOT NULL,
                age INT,
                gender VARCHAR(10),
                diagnosis VARCHAR(100),
                severity VARCHAR(20),
                confidence DECIMAL(5,2),
                notes TEXT,
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE
            ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci
        ''')
        
        # Bảng training_data
        cursor.execute('''
            CREATE TABLE IF NOT EXISTS training_data (
                id INT AUTO_INCREMENT PRIMARY KEY,
                symptoms TEXT NOT NULL,
                age INT,
                gender VARCHAR(10),
                diagnosis VARCHAR(100) NOT NULL,
                severity VARCHAR(20),
                is_verified BOOLEAN DEFAULT 1,
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
            ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci
        ''')
        
        connection.commit()
        
        # Tạo dữ liệu mặc định
        self.create_default_admin()
        self.create_sample_training_data()
        
        cursor.close()
        connection.close()
        print("✅ Các bảng đã được khởi tạo")

    # ========== USER MANAGEMENT ==========
    def authenticate_user(self, username, password):
        """Xác thực user"""
        connection = self.get_connection()
        if not connection:
            return {'success': False, 'error': 'Không thể kết nối database'}
        
        cursor = connection.cursor()
        hashed_pw = hashlib.sha256(password.encode()).hexdigest()
        
        cursor.execute('''
            SELECT id, username, full_name, email, role 
            FROM users WHERE username = %s AND password = %s
        ''', (username, hashed_pw))
        
        user = cursor.fetchone()
        cursor.close()
        connection.close()
        
        if user:
            return {
                'success': True,
                'user': {
                    'id': user[0],
                    'username': user[1],
                    'full_name': user[2],
                    'email': user[3],
                    'role': user[4]
                }
            }
        return {'success': False, 'error': 'Sai tên đăng nhập hoặc mật khẩu'}

import pickle
import numpy as np
from sklearn.naive_bayes import GaussianNB
from sklearn.preprocessing import LabelEncoder
import sys
import os

# Import database
sys.path.append(os.path.dirname(os.path.abspath(__file__)))
from data.database import Database

class ModelTrainer:
    def __init__(self):
        self.db = Database()
        self.model = GaussianNB()
        self.label_encoder = LabelEncoder()
        self.symptom_list = []
        self.model_file = 'trained_model.pkl'
        
    def prepare_data(self):
        """Chuẩn bị dữ liệu từ database"""
        training_data = self.db.get_all_training_data()
        
        if len(training_data) < 2:
            print("❌ Không đủ dữ liệu để train (cần ít nhất 2 mẫu)")
            return None, None
        
        # Lấy danh sách tất cả triệu chứng
        all_symptoms = set()
        for data in training_data:
            all_symptoms.update(data['symptoms'])
        
        self.symptom_list = sorted(list(all_symptoms))
        print(f"📊 Tổng số triệu chứng: {len(self.symptom_list)}")
        
        # Tạo feature vectors
        X = []
        y = []
        
        for data in training_data:
            # Vector triệu chứng (binary encoding)
            feature_vector = [0] * len(self.symptom_list)
            for symptom in data['symptoms']:
                if symptom in self.symptom_list:
                    idx = self.symptom_list.index(symptom)
                    feature_vector[idx] = 1
            
            # Thêm tuổi và giới tính
            feature_vector.append(data['age'])
            feature_vector.append(1 if data['gender'] == 'Nam' else 0)
            
            X.append(feature_vector)
            y.append(data['diagnosis'])
        
        return np.array(X), y
    
    def train(self):
        """Train model và lưu vào file"""
        print("🚀 Bắt đầu train model...")
        
        X, y = self.prepare_data()
        if X is None:
            return False
        
        # Encode labels
        y_encoded = self.label_encoder.fit_transform(y)
        
        # Train model
        self.model.fit(X, y_encoded)
        
        # Lưu model vào file
        model_data = {
            'model': self.model,
            'label_encoder': self.label_encoder,
            'symptom_list': self.symptom_list
        }
        
        with open(self.model_file, 'wb') as f:
            pickle.dump(model_data, f)
        
        print(f"✅ Train thành công!")
        print(f"   - Số mẫu dữ liệu: {len(X)}")
        print(f"   - Số triệu chứng: {len(self.symptom_list)}")
        print(f"   - Số chẩn đoán: {len(self.label_encoder.classes_)}")
        print(f"   - Model đã lưu: {self.model_file}")
        print(f"   - Các chẩn đoán: {', '.join(self.label_encoder.classes_)}")
        
        return True

# Chạy training khi file được execute
if __name__ == '__main__':

import pickle
import numpy as np
import os

class AIAnalyzer:
    def __init__(self):
        self.model_file = 'trained_model.pkl'
        self.model = None
        self.label_encoder = None
        self.symptom_list = []
        self.is_loaded = False
        
        # Load model khi khởi tạo
        self.load_model()
    
    def load_model(self):
        """Load model từ file"""
        if not os.path.exists(self.model_file):
            print(f"⚠️  Chưa có file model: {self.model_file}")
            print("   Vui lòng chạy auto_train_model.py trước!")
            return False
        
        try:
            with open(self.model_file, 'rb') as f:
                model_data = pickle.load(f)
            
            self.model = model_data['model']
            self.label_encoder = model_data['label_encoder']
            self.symptom_list = model_data['symptom_list']
            self.is_loaded = True
            
            print("✅ Load model thành công!")
            print(f"   - Số triệu chứng: {len(self.symptom_list)}")
            print(f"   - Số chẩn đoán: {len(self.label_encoder.classes_)}")
            return True
            
        except Exception as e:
            print(f"❌ Lỗi load model: {e}")
            return False
    
    def predict(self, symptoms, age, gender):
        """Dự đoán chẩn đoán từ triệu chứng"""
        if not self.is_loaded:
            return {
                'success': False,
                'error': 'Model chưa được load. Vui lòng chạy auto_train_model.py trước!'
            }
        
        # Tạo feature vector
        feature_vector = [0] * len(self.symptom_list)
        for symptom in symptoms:
            if symptom in self.symptom_list:
                idx = self.symptom_list.index(symptom)
                feature_vector[idx] = 1
        
        feature_vector.append(age)
        feature_vector.append(1 if gender == 'Nam' else 0)
        
        # Predict
        try:
            prediction = self.model.predict([feature_vector])[0]
            diagnosis = self.label_encoder.inverse_transform([prediction])[0]
            
            # Tính confidence
            probabilities = self.model.predict_proba([feature_vector])[0]
            confidence = max(probabilities) * 100
            
            # Dự đoán severity dựa trên số lượng triệu chứng
            num_symptoms = len(symptoms)
            if num_symptoms <= 2:
                severity = 'Nhẹ'
            elif num_symptoms <= 4:
                severity = 'Trung bình'
            else:
                severity = 'Nặng'
            
            return {
                'success': True,
                'diagnosis': diagnosis,
                'severity': severity,
                'confidence': round(confidence, 2)
            }
        except Exception as e:
            return {
                'success': False,
                'error': f'Lỗi dự đoán: {str(e)}'
            }
    
    def get_available_symptoms(self):
        """Lấy danh sách triệu chứng có thể sử dụng"""
        return self.symptom_list
    
    def get_available_diagnoses(self):
        """Lấy danh sách chẩn đoán"""
        if self.is_loaded:
            return list(self.label_encoder.classes_)
        return []

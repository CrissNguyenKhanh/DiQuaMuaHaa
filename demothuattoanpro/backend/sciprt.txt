# ========== SỬA LẠI ROUTE /api/records ==========

@app.route("/api/records", methods=["POST"])
def add_record():
    """
    Thêm bệnh án mới - CHỈ LƯU AI PREDICTION
    Cần bác sĩ xác nhận sau
    """
    try:
        data = request.json

        # Predict by AI
        prediction = ai.predict(
            symptoms=data["symptoms"], 
            age=data["age"], 
            gender=data["gender"]
        )

        if not prediction["success"]:
            return jsonify(prediction), 400

        # ✅ LƯU RÕ RÀNG: đây là predicted, chưa phải actual
        record_id = db.add_medical_record(
            user_id=data["user_id"],
            symptoms=data["symptoms"],
            age=data["age"],
            gender=data["gender"],
            predicted_diagnosis=prediction["diagnosis"],  # ← AI prediction
            predicted_confidence=prediction["confidence"],
            severity=prediction["severity"],
            actual_diagnosis=None,  # ← Chưa có, cần bác sĩ xác nhận
            confirmed=False,        # ← Chưa xác nhận
            notes=data.get("notes", ""),
        )

        return jsonify({
            "success": True, 
            "record_id": record_id, 
            "prediction": prediction,
            "warning": "⚠️ Đây là chẩn đoán sơ bộ của AI. Vui lòng đến cơ sở y tế để được bác sĩ xác nhận!"
        }), 200
        
    except Exception as e:
        return jsonify({"success": False, "error": str(e)}), 500


# ========== THÊM ROUTE MỚI: XÁC NHẬN CHẨN ĐOÁN ==========

@app.route("/api/records/<int:record_id>/confirm", methods=["PUT"])
def confirm_diagnosis(record_id):
    """
    Bác sĩ xác nhận chẩn đoán thực tế
    CHỈ SAU BƯỚC NÀY thì dữ liệu mới đáng tin cậy
    """
    try:
        data = request.json
        
        # Validate: phải có actual_diagnosis
        if not data.get("actual_diagnosis"):
            return jsonify({
                "success": False, 
                "error": "Vui lòng nhập chẩn đoán thực tế"
            }), 400
        
        success = db.confirm_medical_record(
            record_id=record_id,
            actual_diagnosis=data["actual_diagnosis"],
            confirmed_by=data.get("doctor_id"),
            notes=data.get("notes", "")
        )
        
        if success:
            return jsonify({
                "success": True,
                "message": "✅ Đã xác nhận chẩn đoán!"
            }), 200
        else:
            return jsonify({
                "success": False,
                "error": "Không tìm thấy bệnh án"
            }), 404
        
    except Exception as e:
        return jsonify({"success": False, "error": str(e)}), 500


# ========== SỬA LẠI ROUTE RETRAIN ==========

@app.route("/api/ai/retrain", methods=["POST"])
def retrain():
    """
    ✅ Train model ĐÚNG CÁCH:
    1. Lấy training_data gốc (manual)
    2. Thêm các medical_records ĐÃ XÁC NHẬN
    3. Train
    """
    try:
        # Bước 1: Chuyển confirmed records → training_data
        added = db.add_confirmed_records_to_training()
        print(f"✅ Đã thêm {added} bệnh án đã xác nhận vào training data")
        
        # Bước 2: Train model
        trainer = auto_train_model.ModelTrainer()
        success = trainer.train()

        if success:
            ai.load_model()
            return jsonify({
                "success": True, 
                "message": f"Retrain thành công! Đã thêm {added} bệnh án mới.",
                "details": {
                    "new_confirmed_records": added,
                    "total_training_data": len(db.get_all_training_data())
                }
            }), 200

        return jsonify({
            "success": False, 
            "error": "Không đủ dữ liệu để train"
        }), 400
        
    except Exception as e:
        import traceback
        traceback.print_exc()
        return jsonify({"success": False, "error": str(e)}), 500
